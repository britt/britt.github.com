{{- /* The design system's token layer: every value the site is allowed to use, as CSS custom properties. Concatenated straight out of design/tokens/ (mounted at assets/ds/ — see hugo.toml) rather than transcribed into SCSS, so there is exactly one copy of every value and refreshing the design system cannot leave the site behind. Custom properties, not SCSS variables: they have to survive into the browser so that one declaration serves both colour schemes. fonts.css carries a remote @import and must stay first — @import is only valid before any other rule, and a stylesheet that violates that drops the fonts silently, with no build error. #45 removes the hazard by self-hosting. */ -}} {{- $order := slice "ds/fonts.css" "ds/colors.css" "ds/typography.css" "ds/spacing.css" "ds/layout.css" -}} {{- $parts := slice -}} {{- range $order -}} {{- with resources.Get . -}} {{- $parts = $parts | append . -}} {{- else -}} {{- errorf "design token file not found: %s" . -}} {{- end -}} {{- end -}} {{- $tokens := $parts | resources.Concat "css/tokens.css" -}} {{- if hugo.IsServer -}} {{- else -}} {{- $tokens = $tokens | minify | fingerprint -}} {{- end -}}
