{{- /* Shadows themes/hugo-coder/layouts/partials/head/theme-styles.html. The design system replaces Coder's stylesheet outright; Coder stays on as a template provider. So this emits neither coder.css nor the three Font Awesome woff2 preloads the theme's version shipped (295 KB, for three glyphs — #54 replaces them with inline SVG). Keeping both stylesheets was the original plan (#42) and was abandoned: design/tokens/base.css and Coder's _base.scss style almost the same set of elements, so the two would compete on cascade order, and the winner would be decided by @import sequence rather than by intent. The files are concatenated in the order design/styles.css declares, which is load-bearing — later layers override earlier ones. */ -}} {{- /* Webfonts. The export carries remote @import rules inside fonts.css and playful.css. @import is only valid before any other rule, so the second one is silently dropped by every browser once the files are concatenated — no error, no fonts. They are stripped from the bundle below and re-issued here as tags, which have no ordering constraint. Only the three families the system actually uses are requested. The export's playful.css also asks for Space Grotesk and Instrument Serif, which are referenced by no rule anywhere. #45 self-hosts these and deletes this block. */ -}} {{- $order := slice "ds/fonts.css" "ds/colors.css" "ds/typography.css" "ds/spacing.css" "ds/layout.css" "ds/base.css" "ds/utilities.css" "ds/content.css" "ds/chrome.css" "ds/components.css" "ds/syntax.css" "ds/playful.css" -}} {{- $parts := slice -}} {{- range $order -}} {{- with resources.Get . -}} {{- $parts = $parts | append . -}} {{- else -}} {{- errorf "design system file not found: %s (is design/tokens mounted? see hugo.toml)" . -}} {{- end -}} {{- end -}} {{- $bundle := $parts | resources.Concat "css/site-raw.css" -}} {{- $stripped := $bundle.Content | replaceRE `@import\s+url\(["'][^"']+["']\);?` "" -}} {{- /* No-JS dark mode. The export scopes every dark value to [data-scheme="dark"], an attribute only a script can set. Coder's stylesheet honoured prefers-color-scheme directly, so dropping it would have made dark mode JavaScript-dependent — a regression invisible in any browser with JS on. Rather than write the dark values a second time, the dark rules are lifted out of the bundle and re-emitted with [data-scheme="dark"] rewritten to :not([data-scheme]) inside a prefers-color-scheme query. So: no attribute means the OS decides, in pure CSS with no flash and no script; an explicit attribute always wins, because :not([data-scheme]) stops matching the moment the toggle sets one. Derived, not duplicated — there is still exactly one place a dark colour is written down. */ -}} {{- $darkRules := findRE `[^{}]*\[data-scheme="dark"\][^{}]*\{[^{}]*\}` $stripped -}} {{- if not $darkRules -}} {{- errorf "no [data-scheme=\"dark\"] rules found in the design system bundle — the no-JS dark fallback would be silently empty" -}} {{- end -}} {{- $auto := "" -}} {{- range $darkRules -}} {{- $auto = print $auto (replace . `[data-scheme="dark"]` `:not([data-scheme])`) -}} {{- end -}} {{- $stripped = printf "%s\n@media (prefers-color-scheme:dark){%s}\n" $stripped $auto -}} {{- $styles := resources.FromString "css/site.css" $stripped -}} {{- if hugo.IsServer -}} {{- else -}} {{- $styles = $styles | minify | fingerprint -}} {{- end -}}
